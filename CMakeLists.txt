cmake_minimum_required(VERSION 3.6)

project(freewb)

string(TIMESTAMP FREEWB_VERSION "v4.0 %Y.%m.%d" UTC)
message(STATUS "FREEWB_VERSION: ${FREEWB_VERSION}")

if(${CMAKE_SYSTEM_NAME} MATCHES "Linux")
    set(OS_TYPE "Linux")
    message(STATUS "Detected operating system is : ${OS_TYPE}")
    add_definitions(-D__LINUX__)
elseif(${CMAKE_SYSTEM_NAME} MATCHES "Windows")
    set(OS_TYPE "Windows")
    message(STATUS "Detected operating system is : ${OS_TYPE}")
    add_definitions(-D__WIN__)
else()
    message(STATUS "Unknown operating system.")
endif()

if(${CMAKE_SYSTEM_PROCESSOR} MATCHES "x86_64")
    set(ARCH_TYPE "x86_64")
    message(STATUS "Architecture: ${ARCH_TYPE}")
    add_definitions(-D__X86_64__)
elseif(${CMAKE_SYSTEM_PROCESSOR} MATCHES "aarch64")
    set(ARCH_TYPE "aarch64")
    message(STATUS "Architecture: ${ARCH_TYPE}")
    add_definitions(-D__AARCH64__)
elseif(${CMAKE_SYSTEM_PROCESSOR} MATCHES "loongarch64")
    message(STATUS "debian env : $ENV{DEB_HOST_ARCH}")
    if ($ENV{DEB_HOST_ARCH} STREQUAL "loong64")
	    set(ARCH_TYPE "loong64")
	    message(STATUS "Architecture: ${ARCH_TYPE}")
	    add_definitions(-D__LOONG64__)
    else()
	    set(ARCH_TYPE "loongarch64")
	    message(STATUS "Architecture: ${ARCH_TYPE}")
	    add_definitions(-D__LOONGARCH64__)
    endif()
elseif(${CMAKE_SYSTEM_PROCESSOR} MATCHES "mips64el")
    set(ARCH_TYPE "mips64el")
    message(STATUS "Architecture: ${ARCH_TYPE}")
    add_definitions(-D__MIPS64__)
elseif(${CMAKE_SYSTEM_PROCESSOR} MATCHES "riscv64")
    set(ARCH_TYPE "riscv64")
    message(STATUS "Architecture: ${ARCH_TYPE}")
    add_definitions(-D__RISCV64__)
elseif(${CMAKE_SYSTEM_PROCESSOR} MATCHES "sw_64")
    set(ARCH_TYPE "sw_64")
    message(STATUS "Architecture: ${ARCH_TYPE}")
    add_definitions(-D__SW_64__)
else()
    message(STATUS "Unknown architecture.")
endif()

option(ENABLE_FCITX4 "enable support fcitx4" Off)
option(ENABLE_FCITX5 "enable support fcitx5" Off)
option(ENABLE_TEST "enable freewb test cases" Off)
option(ENABLE_UKUI "enable support ukui" Off)

if (${OS_TYPE} STREQUAL "Linux")
    find_package(ECM REQUIRED 1.0.0)
    set(CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake" ${ECM_MODULE_PATH} ${CMAKE_MODULE_PATH})
    include(FindPkgConfig)
    include(GNUInstallDirs)
    include(ECMUninstallTarget)
endif()

# Hide spdlog symbols in shared libraries to avoid ABI conflicts when another
# DSO embeds a different spdlog version in the same process.
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -Wl,--version-script=${CMAKE_SOURCE_DIR}/cmake/hide-spdlog-exports.map")

set(FREEWB_INSTALL_PREFIX ${CMAKE_INSTALL_PREFIX})
set(FREEWB_INSTALL_INCLUDEDIR ${CMAKE_INSTALL_FULL_INCLUDEDIR})
set(FREEWB_INSTALL_LIBDIR ${CMAKE_INSTALL_FULL_LIBDIR}/freewb)
set(FREEWB_INSTALL_DATADIR ${CMAKE_INSTALL_FULL_DATADIR})
set(FREEWB_INSTALL_PKGDATADIR ${CMAKE_INSTALL_FULL_DATADIR}/freewb)
set(FREEWB_INSTALL_BINDIR ${CMAKE_INSTALL_FULL_BINDIR})
set(FREEWB_INSTALL_LOCALEDIR ${CMAKE_INSTALL_FULL_DATADIR}/locale)

message(STATUS "FREEWB_INSTALL_PREFIX: ${FREEWB_INSTALL_PREFIX}")
message(STATUS "FREEWB_INSTALL_INCLUDEDIR: ${FREEWB_INSTALL_INCLUDEDIR}")
message(STATUS "FREEWB_INSTALL_LIBDIR: ${FREEWB_INSTALL_LIBDIR}")
message(STATUS "FREEWB_INSTALL_DATADIR: ${FREEWB_INSTALL_DATADIR}")
message(STATUS "FREEWB_INSTALL_PKGDATADIR: ${FREEWB_INSTALL_PKGDATADIR}")
message(STATUS "FREEWB_INSTALL_BINDIR: ${FREEWB_INSTALL_BINDIR}")
message(STATUS "FREEWB_INSTALL_LOCALEDIR: ${FREEWB_INSTALL_LOCALEDIR}")
 
configure_file(config.h.in ${CMAKE_SOURCE_DIR}/include/config.h IMMEDIATE @ONLY)

add_subdirectory(src)
add_subdirectory(data)
add_subdirectory(help)
add_subdirectory(resources)
add_subdirectory(po)
if(ENABLE_TEST)
    enable_testing()
    add_subdirectory(test)
endif()
